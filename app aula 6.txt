import React from 'react';
import { View, Text, Button, StyleSheet } from 'react-native';
import { NavigationContainer } from '@react-navigation/native';
import { createNativeStackNavigator } from '@react-navigation/native-stack';

const Stack = createNativeStackNavigator();

function HomeScreen({ navigation }) {
  return (
    <View style={styles.container}>
      <Text style={styles.titulo}>Tela Inicial</Text>

      <Button
        title="Ir para Detalhes"
        onPress={() => navigation.navigate('Detalhes')}
      />
    </View>
  );
}

function DetailScreen({ navigation }) {
  return (
    <View style={styles.container}>
      <Text style={styles.titulo}>Tela de Detalhes</Text>

      <Button
        title="Navigate"
        onPress={() => navigation.navigate('Detalhes')}
      />

      <Button
        title="Push"
        onPress={() => navigation.push('Detalhes')}
      />

      <Button
        title="GoBack"
        onPress={() => navigation.goBack()}
      />

      <Button
        title="PopToTop"
        onPress={() => navigation.popToTop()}
      />

      <Button
        title="Replace"
        onPress={() => navigation.replace('OutraTela')}
      />
    </View>
  );
}

function OutraTela() {
  return (
    <View style={styles.container}>
      <Text style={styles.titulo}>Outra Tela</Text>
    </View>
  );
}

export default function App() {
  return (
    <NavigationContainer>
      <Stack.Navigator>
        <Stack.Screen
          name="Inicio"
          component={HomeScreen}
        />

        <Stack.Screen
          name="Detalhes"
          component={DetailScreen}
        />

        <Stack.Screen
          name="OutraTela"
          component={OutraTela}
        />
      </Stack.Navigator>
    </NavigationContainer>
  );
}

const styles = StyleSheet.create({
  container: {
    flex: 1,
    justifyContent: 'center',
    padding: 20,
    gap: 15,
    backgroundColor: '#fff'
  },
  titulo: {
    fontSize: 22,
    fontWeight: 'bold',
    textAlign: 'center'
  }
});

Pergunta: O navigate() vai para outra tela, permitindo voltar para a anterior. Já o replace() substitui a tela atual por outra, sem permitir voltar para a tela substituída.

1. O contador aumenta apenas uma vez porque as três atualizações usam o mesmo valor antigo.

2. Para aumentar três vezes, usamos setContador(valor => valor + 1) em cada atualização.

3. Não devemos colocar async diretamente na função do useEffect, porque ela precisa retornar uma função de limpeza ou nada.

4. A correção é criar uma função async dentro do useEffect e chamá-la.
